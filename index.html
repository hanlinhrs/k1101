🛠️ 終極排查與設定指引：

第一步：確認您 Google 試算表上的工作表名稱

請確保您的試算表左下角包含完全一致的兩個分頁：

  • 📁 「員工清單」 (A1 儲存格首行為：姓名)
  • 📁 「紀錄列表」 (A1 儲存格首行為：日期、填寫人、員工姓名、狀況說明、備註)

第二步：使用對齊的正式 Apps Script 程式碼

請完整複製下方代碼，覆蓋覆蓋貼上您的 Google Apps Script 編輯器，存檔後進行部署：

const SPREADSHEET_ID = "1eXM9urtZEpKWKJHhxVbjrQHcjU6g_JYl0yjeUa-qw5c";

// 處理 GET 請求：同時讀取紀錄列表與員工清單
function doGet(e) {
  try {
    const action = e ? e.parameter.action : "";
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    
    if (action === "getRecords") {
      // 1. 讀取紀錄列表
      const recordSheet = ss.getSheetByName("紀錄列表");
      if (!recordSheet) {
        throw new Error("找不到名為 '紀錄列表' 的工作表分頁，請確認您的試算表分頁名稱是否正確。");
      }
      
      const records = [];
      const recordRows = recordSheet.getDataRange().getValues();
      if (recordRows.length > 1) {
        const headers = recordRows[0];
        for (let i = 1; i < recordRows.length; i++) {
          let row = recordRows[i];
          let record = {};
          for (let j = 0; j < headers.length; j++) {
            let key = headers[j].toString().trim();
            if (key === "日期" || key === "date") record.date = row[j];
            else if (key === "填寫人" || key === "recorder") record.recorder = row[j];
            else if (key === "員工姓名" || key === "員工" || key === "name") record.name = row[j];
            else if (key === "狀況說明" || key === "desc") record.desc = row[j];
            else if (key === "備註" || key === "note") record.note = row[j];
            else record[key] = row[j];
          }
          records.push(record);
        }
      }
      
      // 2. 讀取員工清單
      const empSheet = ss.getSheetByName("員工清單");
      if (!empSheet) {
        throw new Error("找不到名為 '員工清單' 的工作表分頁，請確認您的試算表分頁名稱是否正確。");
      }
      
      const employees = [];
      const empRows = empSheet.getDataRange().getValues();
      for (let k = 1; k < empRows.length; k++) {
        let empName = empRows[k][0];
        if (empName && empName.toString().trim() !== "") {
          employees.push(empName.toString().trim());
        }
      }
      
      const result = {
        success: true,
        records: records,
        employees: employees
      };
      
      return ContentService.createTextOutput(JSON.stringify(result))
        .setMimeType(ContentService.MimeType.JSON)
        .setHeader('Access-Control-Allow-Origin', '*');
        
    } else {
      return ContentService.createTextOutput(JSON.stringify({
        success: false,
        error: "未知的 action 指令，請嘗試呼叫 ?action=getRecords"
      }))
        .setMimeType(ContentService.MimeType.JSON)
        .setHeader('Access-Control-Allow-Origin', '*');
    }
    
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      success: false,
      error: error.toString()
    }))
      .setMimeType(ContentService.MimeType.JSON)
      .setHeader('Access-Control-Allow-Origin', '*');
  }
}

// 處理 POST 請求
function doPost(e) {
  try {
    const postData = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const recordSheet = ss.getSheetByName("紀錄列表");
    
    if (!recordSheet) {
      throw new Error("找不到名為 '紀錄列表' 的工作表分頁。");
    }
    
    if (postData.action === "addRecord") {
      recordSheet.appendRow([
        postData.date,
        postData.recorder,
        postData.name,
        postData.desc,
        postData.note
      ]);
      
      return ContentService.createTextOutput(JSON.stringify({ success: true }))
        .setMimeType(ContentService.MimeType.JSON)
        .setHeader('Access-Control-Allow-Origin', '*');
    }
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, message: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON)
      .setHeader('Access-Control-Allow-Origin', '*');
  }
}

第三步：正確部署並獲取 /exec 網址

  1. 在 Apps Script 點擊右上角 「部署 (Deploy)」➡️「管理部署」➡️「編輯」（或 「新建部署」）。
  2. 設定正確的權限值（核心關鍵）：
    • 將應用程式執行為 (Execute as)：選取 「我 (Me)」
    • 誰有權限存取 (Who has access)：務必選取 「任何人 (Anyone)」
  3. 點擊「部署」。初次部署時，請務必點選「授予存取權」完成 Google 安全授權核可。
